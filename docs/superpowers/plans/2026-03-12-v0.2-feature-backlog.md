import { describe, expect, test } from "bun:test";
import { buildScores } from "../engine/scoreboard";
import type { AgentState } from "../types";

function makeAgent(overrides: Partial<AgentState> & { id: string; name: string; color: string }): AgentState {
  return {
    identity_dna: {
      archetype: "Test",
      core_values: ["test"],
      speech_style: "calm",
      risk_appetite: 50,
      aggression_bias: 50,
      diplomacy_bias: 50,
      creativity_bias: 50,
      signature_moves: ["test"],
      taboos: []
    },
    goal_weights: { territory: 0.25, art: 0.25, revenge: 0.25, reputation: 0.25 },
    emotion: { anger: 20, fear: 20, confidence: 50, satisfaction: 50 },
    reputation: 70,
    energy: 3,
    cooldowns: { burst: 0 },
    relations: [],
    memory: [],
    ...overrides
  };
}

function makeBoard(width: number, height: number, owners: Array<{ x: number; y: number; owner: string; color: string }>) {
  const board = Array.from({ length: height }, () =>
    Array.from({ length: width }, () => ({ owner: null as string | null, color: "#111111" }))
  );
  for (const { x, y, owner, color } of owners) {
    board[y][x] = { owner, color };
  }
  return board;
}

describe("buildScores", () => {
  test("returns scores sorted by final_score descending", () => {
    const agents = [
      makeAgent({ id: "a1", name: "Alpha", color: "#FF0000" }),
      makeAgent({ id: "a2", name: "Beta", color: "#00FF00" })
    ];
    const board = makeBoard(8, 8, [
      { x: 0, y: 0, owner: "a1", color: "#FF0000" },
      { x: 1, y: 0, owner: "a1", color: "#FF0000" },
      { x: 2, y: 0, owner: "a2", color: "#00FF00" }
    ]);

    const scores = buildScores({
      agents,
      board,
      width: 8,
      height: 8,
      mythAestheticScore: () => 50
    });

    expect(scores.length).toBe(2);
    expect(scores[0].final_score).toBeGreaterThanOrEqual(scores[1].final_score);
    expect(scores[0].agent_id).toBe("a1");
    expect(scores[0].territory_cells).toBe(2);
    expect(scores[1].territory_cells).toBe(1);
  });

  test("handles zero territory", () => {
    const agents = [makeAgent({ id: "a1", name: "Alpha", color: "#FF0000" })];
    const board = makeBoard(4, 4, []);

    const scores = buildScores({
      agents,
      board,
      width: 4,
      height: 4,
      mythAestheticScore: () => 0
    });

    expect(scores[0].territory_cells).toBe(0);
    expect(scores[0].territory_score).toBe(0);
  });

  test("art_score varies with mythAestheticScore", () => {
    const agents = [makeAgent({ id: "a1", name: "Alpha", color: "#FF0000" })];
    const board = makeBoard(4, 4, [{ x: 0, y: 0, owner: "a1", color: "#FF0000" }]);

    const lowArt = buildScores({ agents, board, width: 4, height: 4, mythAestheticScore: () => 0 });
    const highArt = buildScores({ agents, board, width: 4, height: 4, mythAestheticScore: () => 100 });

    expect(highArt[0].art_score).toBeGreaterThan(lowArt[0].art_score);
  });
});

import { describe, expect, test } from "bun:test";
import { normalizeTurnDecision } from "../decisionNormalizer";

const defaults = { agentId: "a1", round: 1, color: "#FF0000" };

describe("normalizeTurnDecision", () => {
  test("normalizes valid complete input", () => {
    const raw = {
      agent_id: "a1",
      round: 1,
      intent: "expand",
      public_message: "forward",
      private_messages: [],
      treaty_proposals: [],
      actions: [{ action: "paint", x: 5, y: 3, color: "#00FF00" }],
      emotion_delta: { anger: 2, fear: -1, confidence: 3, satisfaction: 1 },
      mood_change_reason: "moving ahead"
    };

    const result = normalizeTurnDecision({ raw, ...defaults });
    expect(result.agent_id).toBe("a1");
    expect(result.actions[0].action).toBe("paint");
  });

  test("falls back to wait on empty input", () => {
    const result = normalizeTurnDecision({ raw: {}, ...defaults });
    expect(result.actions).toEqual([{ action: "wait" }]);
    expect(result.agent_id).toBe("a1");
  });

  test("normalizes action name aliases", () => {
    const raw = { actions: [{ action: "attack", x: 1, y: 2 }] };
    const result = normalizeTurnDecision({ raw, ...defaults });
    expect(result.actions[0].action).toBe("invade");
  });

  test("normalizes treaty type aliases", () => {
    const raw = {
      treaty_proposals: [{ target_id: "a2", type: "joint-attack", duration_rounds: 2 }]
    };
    const result = normalizeTurnDecision({ raw, ...defaults });
    expect(result.treaty_proposals[0].type).toBe("joint_attack");
  });

  test("caps actions at 3", () => {
    const raw = {
      actions: [
        { action: "paint", x: 0, y: 0, color: "#FF0000" },
        { action: "paint", x: 1, y: 1, color: "#FF0000" },
        { action: "paint", x: 2, y: 2, color: "#FF0000" },
        { action: "paint", x: 3, y: 3, color: "#FF0000" }
      ]
    };
    const result = normalizeTurnDecision({ raw, ...defaults });
    expect(result.actions.length).toBe(3);
  });

  test("derives intent from actions when not explicit", () => {
    const raw = { actions: [{ action: "invade", x: 1, y: 2 }] };
    const result = normalizeTurnDecision({ raw, ...defaults });
    expect(result.intent).toBe("revenge");
  });

  test("clamps emotion_delta values to [-20, 20]", () => {
    const raw = {
      emotion_delta: { anger: 100, fear: -100, confidence: 50, satisfaction: -50 }
    };
    const result = normalizeTurnDecision({ raw, ...defaults });
    expect(result.emotion_delta.anger).toBeLessThanOrEqual(20);
    expect(result.emotion_delta.fear).toBeGreaterThanOrEqual(-20);
  });

  test("handles string coordinates", () => {
    const raw = { actions: [{ action: "paint", target: "4,7", color: "#FF0000" }] };
    const result = normalizeTurnDecision({ raw, ...defaults });
    expect(result.actions[0]).toEqual({ action: "paint", x: 4, y: 7, color: "#FF0000" });
  });

  test("handles row/col format", () => {
    const raw = { actions: [{ action: "fortify", row: 3, col: 5 }] };
    const result = normalizeTurnDecision({ raw, ...defaults });
    expect(result.actions[0]).toEqual({ action: "fortify", x: 5, y: 3 });
  });
});

```ts
function replayList(query?: { mode?: string; agents_min?: number; agents_max?: number }) {
  if (!fs.existsSync(outputRoot)) return [];
  const entries = fs
    .readdirSync(outputRoot)
    .filter((name) => /^replay-.*\.json$/.test(name))
    .map((name) => {
      const fullPath = path.join(outputRoot, name);
      const stat = fs.statSync(fullPath);

      let mode: "dry-run" | "live" = "live";
      let agentCount = 0;
      try {
        const fd = fs.openSync(fullPath, "r");
        const buf = Buffer.alloc(512);
        fs.readSync(fd, buf, 0, 512, 0);
        fs.closeSync(fd);
        const head = buf.toString("utf8");
        if (/"dry_run"\s*:\s*true/.test(head)) mode = "dry-run";
        const agentMatch = head.match(/"agent_count"\s*:\s*(\d+)/);
        if (agentMatch) agentCount = Number(agentMatch[1]);
      } catch {
        // ignore read errors
      }

      return {
        name,
        mtime: new Date(stat.mtimeMs).toISOString(),
        bytes: stat.size,
        mode,
        agent_count: agentCount
      };
    })
    .sort((a, b) => Date.parse(b.mtime) - Date.parse(a.mtime));

  return entries.filter((item) => {
    if (query?.mode && item.mode !== query.mode) return false;
    if (query?.agents_min && item.agent_count < query.agents_min) return false;
    if (query?.agents_max && item.agent_count > query.agents_max) return false;
    return true;
  });
}
```
